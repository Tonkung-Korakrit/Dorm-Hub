## วิธีติดตั้ง
npm init playwright@latest

## เขียนเพื่อทำให้กด แล้วจะเจนเป็นโค้ด:
npx playwright codegen https://tudormbooking-production.up.railway.app/
npx playwright codegen https://tudormbooking-production.up.railway.app/ -o tester_code/e2e/new_booking.spec.ts

## รันแบบไม่เห็นหน้าจอ (เร็ว):
npx playwright test

## รันแบบเห็นหน้าต่าง UI (แนะนำตอนเริ่มเขียน):
npx playwright test --ui

npx playwright test tester_code/e2e/booking.spec.ts --ui
